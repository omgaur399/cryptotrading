4352: function updateReplayAnalyticsPanel(chartId, openPnl = 0) {
4353:     const chartData = state.charts[chartId];
4354:     if (!chartData || !chartData.replay) return;
4355:     
4356:     const history = chartData.replay.paper.history;
4357:     const wins = history.filter(t => t.pnl > 0);
4358:     const losses = history.filter(t => t.pnl <= 0);
4359:     
4360:     const winRate = history.length > 0 ? ((wins.length / history.length) * 100).toFixed(1) : '0.0';
4361:     
4362:     let grossWin = 0;
4363:     let grossLoss = 0;
4364:     let largestWin = 0;
4365:     let largestLoss = 0;
4366:     
4367:     wins.forEach(w => { grossWin += w.pnl; if (w.pnl > largestWin) largestWin = w.pnl; });
4368:     losses.forEach(l => { grossLoss += Math.abs(l.pnl); if (l.pnl < largestLoss) largestLoss = l.pnl; });
4369:     
4370:     const avgWin = wins.length > 0 ? (grossWin / wins.length) : 0;
4371:     const avgLoss = losses.length > 0 ? (grossLoss / losses.length) : 0;
4372:     const profitFactor = grossLoss > 0 ? (grossWin / grossLoss).toFixed(2) : (grossWin > 0 ? '∞' : '0.00');
4373:     
4374:     const netPnl = (chartData.replay.paper.balance - chartData.replay.paper.initialBalance) + openPnl;
4375:     const currentEquity = chartData.replay.paper.balance + openPnl;
4376:     
4377:     window.ModalService.renderReplayAnalyticsPanel(
4378:         document.getElementById(`${chartId}-container`),
4379:         {
4380:             panelId: `replay-analytics-panel-${chartId}`,
4381:             initialBalance: chartData.replay.paper.initialBalance,
4382:             currentEquity: currentEquity,
4383:             netPnl: netPnl,
4384:             winRate: winRate,
4385:             totalTrades: history.length,
4386:             avgWin: avgWin,
4387:             avgLoss: avgLoss,
4388:             largestWin: largestWin,
4389:             largestLoss: largestLoss,
4390:             profitFactor: profitFactor
4391:         }
